import bpy

# Start from an empty scene
bpy.ops.scene.new(type='NEW')

# Set the scene name
bpy.context.scene.name = 'Shot'

# Set the frame range
bpy.context.scene.frame_start = 1
bpy.context.scene.frame_end = 124

# Set the frame rate
bpy.context.scene.render.fps = 24

# Create a camera
cam = bpy.data.cameras.new(name="Wide Moving Shot")
cam.sensor_width = 32.0
cam.sensor_height = 24.0
cam.type = 'PANO'
cam.pano_type = 'RECTANGULAR'
cam.angle = 1.0
cam.name = 'Wide Moving Shot'

# Add camera to active scene
scene = bpy.context.scene
scene.camera = cam

# Create a light
lamp_data = bpy.data.lamps.new(name="Main Lamp", type='IES')
lamp_data.energy = 1.0
lamp_data.use_diffuse = True
lamp_data.use_specular = True
lamp_data.shadow_soft_size = 1.0
lamp_data.type = 'IES'
lamp_data.shadow_method = 'SOLID'
lamp = bpy.data.objects.new(name="Main Lamp", object_data=lamp_data)
lamp.scale[0] = 0.5
lamp.scale[1] = 0.5
lamp.scale[2] = 0.5
scene.collection.objects.link(lamp)

# Create geometry
# James
bpy.ops.mesh.primitive_cube_add(size=1, location=(-2.5, 0, 2.5))
bpy.context.object.name = 'James'
bpy.context.object.data.materials.append(bpy.data.materials.new(name="James Material"))
bpy.context.object.data.materials[0].diffuse_color = (0.1, 0.4, 0.8, 1.0)
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.select_all(action='SELECT')
bpy.ops.mesh.loopcut_slide(orient_axis='Y', MARGIN=0.2)
bpy.ops.object.mode_set(mode='OBJECT')
bpy.ops.object.modifier_add(type='SUBSURFACE')
bpy.context.object.modifiers["Subsurface"].level = 2
